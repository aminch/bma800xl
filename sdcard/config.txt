#
# BMA800XL - bare metal Atari 800XL emulator for the Raspberry Pi 3
#
# Raspberry Pi firmware settings. Emulator options are in cmdline.txt.
#

# BMA800XL runs in 32 bit mode.
arm_64bit=0
kernel=kernel8-32.img

# Video: 1280x720 at 50 Hz, the right rate for a PAL Atari (the default).
# For an NTSC Atari use 60 Hz instead: hdmi_mode=4
# The emulator's frames are locked to the display when its refresh rate
# matches the Atari's, which gives perfectly smooth scrolling.
disable_overscan=1
hdmi_group=1
hdmi_mode=19

# Other useful modes (hdmi_group=1):
#   hdmi_mode=19  720p  50 Hz        hdmi_mode=4   720p  60 Hz
#   hdmi_mode=31  1080p 50 Hz        hdmi_mode=16  1080p 60 Hz

# Uncomment if you get no picture
#hdmi_safe=1

# Uncomment to force HDMI (rather than DVI) mode, for sound over HDMI on
# some monitors
#hdmi_drive=2

# Uncomment to boost the HDMI signal if you get interference
#config_hdmi_boost=4

# Composite video (the 3.5mm jack) instead of HDMI: sdtv_mode=2 is PAL,
# sdtv_mode=0 is NTSC. Unplug HDMI to use it.
sdtv_mode=2

# Scaling filter: 8 = sharp pixels, 2 = soft, 6 = softer (default)
#scaling_kernel=8

gpu_mem=64
disable_splash=1
boot_delay=0
